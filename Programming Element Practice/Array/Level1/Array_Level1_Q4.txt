//Array_Level1_Q4
import java.util.*;
class Level1_Q4 {
    public static void main(String[] args) {
        Scanner sc=new Scanner(System.in);
        double[] arr=new double[10];
        double total=0.0;
        int index=0;
        while(true){
            double num=sc.nextDouble();
            if(num<=0){
                break;
            }
            if (index==10){
            break;
            }
        arr[index]=num;
        index++;
        }
        for(int i=0; i<index; i++){
        total+=arr[i];
    }
    System.out.print("Sum is "+total);
    }
}
